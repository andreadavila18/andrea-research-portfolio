Quantitative Results: Collective Action and Violence Prevention

Overview

This section presents the quantitative findings from my dissertation, focusing on how agricultural cooperative membership and peasant associations relate to conflict-related victimization in rural Colombia. I use panel data (2016–2022) at the municipality level to examine long-term patterns of collective action and their association with violence.

The analysis includes a Between Effects regression model, which captures how average levels of cooperative membership across municipalities relate to average levels of victimization over time.

# Prepare data
panel_data1 <- panel_data %>% filter(year >= 2016 & year <= 2021)
panel_data2 <- panel_data %>% filter(year >= 2018 & year <= 2022)

panel_data1 <- pdata.frame(panel_data1, index = c("mun", "year"))
panel_data2 <- pdata.frame(panel_data2, index = c("mun", "year"))

panel_data1 <- panel_data1 %>% mutate(PDET = as.numeric(PDET))

panel_data1_BE <- panel_data1 %>%
  group_by(mun) %>%
  summarise(PDET = first(PDET), across(where(is.numeric), mean, na.rm = TRUE))

# Estimate Between Effects Models
BE_uariv_members <- lm(log_victims ~ log_rate_members, data = panel_data1_BE)
BE_uariv_pa <- lm(log_victims ~ log_rate_pa_active_imp, data = panel_data2_BE)

BE_uariv_members_c <- lm(log_victims ~ log_rate_members + PDET + PDET:log_rate_members + gestion + educacion + 
                           salud + servicios + log_added_value +  
                           community_councils_cero + indigenous_reserves_cero, 
                         data = panel_data1_BE)

BE_uariv_pa_c <- lm(log_victims ~ log_rate_pa_active_imp + PDET + PDET:log_rate_pa_active_imp + gestion + educacion + 
                      salud + servicios + log_added_value  + 
                      community_councils_cero + indigenous_reserves_cero, 
                    data = panel_data2_BE)

# Robust SEs
robust_se_BE_uariv_members <- vcovHC(BE_uariv_members, type = "HC0")
robust_se_BE_uariv_pa <- vcovHC(BE_uariv_pa, type = "HC0")
robust_se_BE_uariv_members_c <- vcovHC(BE_uariv_members_c, type = "HC0")
robust_se_BE_uariv_pa_c <- vcovHC(BE_uariv_pa_c, type = "HC0")

coeftest_BE_uariv_members <- coeftest(BE_uariv_members, vcov = robust_se_BE_uariv_members)
coeftest_BE_uariv_pa <- coeftest(BE_uariv_pa, vcov = robust_se_BE_uariv_pa)
coeftest_BE_uariv_members_c <- coeftest(BE_uariv_members_c, vcov = robust_se_BE_uariv_members_c)
coeftest_BE_uariv_pa_c <- coeftest(BE_uariv_pa_c, vcov = robust_se_BE_uariv_pa_c)

# Export results
stargazer(BE_uariv_members, BE_uariv_pa, BE_uariv_members_c, BE_uariv_pa_c,
  se = list(
    sqrt(diag(robust_se_BE_uariv_members)), 
    sqrt(diag(robust_se_BE_uariv_pa)), 
    sqrt(diag(robust_se_BE_uariv_members_c)), 
    sqrt(diag(robust_se_BE_uariv_pa_c))
  ),
  type = "html",
  title = "Between Effects Models with Robust Standard Errors",
  dep.var.labels = "Victims in Conflict - UARIV",
  add.lines = list(
    c("Observations", 
      nobs(BE_uariv_members), 
      nobs(BE_uariv_pa), 
      nobs(BE_uariv_members_c), 
      nobs(BE_uariv_pa_c))
  )
)
